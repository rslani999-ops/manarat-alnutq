// ============================================================ // ====== صفحة الطالب (بدون تكرار) ====== // ============================================================ async function renderStudentMenu(app) { renderTopbar(app, 'صفحة الطالب والجلسات', 'متابعة مستوى النطق والتقييمات'); // ===== حاوية رئيسية واحدة للمحتوى ===== let container = document.getElementById('studentContent'); if (!container) { container = document.createElement('div'); container.id = 'studentContent'; app.appendChild(container); } else { container.innerHTML = ''; // إفراغ الحاوية بالكامل } // جلب بيانات الطالب const uDoc = await getDoc(doc(db, "users", state.user.uid)); const studentData = uDoc.data() || {}; const diag = studentData.diagnostic || {}; const passed = ALL_LETTERS.filter(l => diag[l] === 'passed'); const need = ALL_LETTERS.filter(l => diag[l] === 'need' || diag[l] === 'unclear'); // ===== 1. ملخص المستوى ===== const summary = document.createElement('div'); summary.className = 'card'; summary.innerHTML = `

📊 ملخص مستوى الطالب والنطق

الحروف المتقنة: ${passed.length ? passed.join(' - ') : 'لا يوجد'}
حروف تحتاج تدريب: ${need.length ? need.join(' - ') : 'لا يوجد'}
`; container.appendChild(summary); // ===== 2. الخطة الفردية (IEP) ===== const iepCard = document.createElement('div'); iepCard.className = 'card'; iepCard.innerHTML = `

📋 الخطة الفردية (IEP)

`; try { const iepDoc = await getDoc(doc(db, "iep", state.user.uid)); if (iepDoc.exists()) { const iepData = iepDoc.data(); iepCard.innerHTML += `
📝 تشخيص الحالة:
${iepData.diagnosis || 'لا يوجد تشخيص'}
🎯 الهدف العام:
${iepData.longGoal || 'لا يوجد هدف عام'}
📌 الأهداف القصيرة:
${iepData.shortGoals || 'لا توجد أهداف قصيرة'}
💡 التوصيات:
${iepData.notes || 'لا توجد توصيات'}
آخر تحديث: ${iepData.updatedAt ? new Date(iepData.updatedAt).toLocaleDateString('ar-SA') : 'غير معروف'}

ℹ️ هذه الخطة للقراءة فقط، يمكنك مناقشتها مع معلمك.

`; } else { iepCard.innerHTML += `

⚠️ لا توجد خطة فردية مسجلة حتى الآن. يرجى التواصل مع معلمك لإجراء التشخيص.

`; } } catch(e) { console.error('خطأ في جلب الخطة:', e); iepCard.innerHTML += `

⚠️ خطأ في تحميل الخطة الفردية.

`; } container.appendChild(iepCard); // ===== 3. سجل الجلسات ===== const sessionsCard = document.createElement('div'); sessionsCard.className = 'card'; sessionsCard.innerHTML = '

📂 سجل جلسات التدريب الخاصة بك

'; try { const qS = query(collection(db, "sessions"), where("studentId", "==", state.user.uid)); const snap = await getDocs(qS); if (snap.empty) { sessionsCard.innerHTML += '

لا توجد جلسات مسجلة.

'; } else { const allS = []; snap.forEach(d => allS.push({ id: d.id, ...d.data() })); allS.sort((a,b) => (a.sessionNumber||0) - (b.sessionNumber||0)); allS.forEach(sess => { const typeInfo = SESSION_TYPES.find(t => t.id === sess.sessionType) || SESSION_TYPES[0]; const sItem = document.createElement('div'); sItem.className = 'session-card'; sItem.innerHTML = `
${typeInfo.icon} ${typeInfo.name} حرف ${sess.letter} - #${sess.sessionNumber || '?'} - ${sess.date}
${sess.goal}
`; sessionsCard.appendChild(sItem); setTimeout(() => { document.getElementById(`openSess_${sess.id}`).onclick = () => { state.currentSession = { id: sess.id, ...sess }; state.view = 'single-session'; render(); }; }, 0); }); } } catch(e) { sessionsCard.innerHTML += `

⚠️ خطأ في تحميل الجلسات: ${e.message}

`; } container.appendChild(sessionsCard); }
